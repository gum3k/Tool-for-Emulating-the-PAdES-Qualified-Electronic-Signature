59def decrypt_private_key(pin):
60 global private_key
61 if not key_file_path:
62 messagebox.showerror("Błąd", "Najpierw wykryj klucz.")
63 return False
64 try:
65 with open(key_file_path, "rb") as f:
66 salt = f.read(16)
67 nonce = f.read(16)
68 tag = f.read(16)
69 ciphertext = f.read()
70 key = scrypt(pin.encode(), salt, 32, N=2**14, r=8, p=1)
71 cipher = AES.new(key, AES.MODE_GCM, nonce=nonce)
72 decrypted = cipher.decrypt_and_verify(ciphertext, tag)
73 private_key = RSA.import_key(decrypted)
74 return True
75 except Exception as e:
76 messagebox.showerror("Błąd", f"Nie udało się odszyfrować klucza:\n{e}")
77 return False

95def sign_pdf_embed():
96 global pdf_path, private_key
97 if not private_key or not pdf_path:
98 messagebox.showerror("Błąd", "Brak klucza prywatnego lub pliku PDF.")
99 return
100 try:
101 reader = PdfReader(pdf_path)
102 writer = PdfWriter()
103
104 for page in reader.pages:
105 writer.add_page(page)
106
107 data = b"".join([page.extract_text().encode() for page in reader.pages])
108
109 h = SHA256.new(data)
110 signature = pkcs1_15.new(private_key).sign(h)
111
112 writer.add_metadata({
113 "/Signature": signature.hex(),
114 })
115
116 signed_path = pdf_path.replace(".pdf", "_signed.pdf")
117 with open(signed_path, "wb") as f:
118 writer.write(f)
119
120 messagebox.showinfo("Sukces", f"PDF podpisany jako: {signed_path}")
121 except Exception as e:
122 messagebox.showerror("Błąd", f"Podpisanie nie powiodło się:\n{e}")

156def verify_pdf_signature():
157 global signed_pdf_path, public_key
158 if not signed_pdf_path or not public_key:
159 messagebox.showerror("Błąd", "Brak pliku PDF lub klucza publicznego.")
160 return
161 try:
162 reader = PdfReader(signed_pdf_path)
163 signature_hex = reader.metadata.get("/Signature", "")
164
165 if not signature_hex:
166 messagebox.showerror("Błąd", "Brak podpisu w pliku PDF.")
167 return
168 signature = bytes.fromhex(signature_hex)
169 neutralized_data = b"".join([page.extract_text().encode() for page in reader.pages])
170 if len(signature) != 512:
171 messagebox.showerror("Błąd", "Nieprawidłowy podpis. Oczekiwano 512 bajtów.")
172 return
173 h = SHA256.new(neutralized_data)
174 pkcs1_15.new(public_key).verify(h, signature)
175 messagebox.showinfo("Weryfikacja", "Podpis poprawny.")
176 except Exception as e:
177 messagebox.showerror("Weryfikacja", f"Podpis nieprawidłowy lub błąd:\n{e}")
