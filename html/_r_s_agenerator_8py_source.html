27def generate_rsa_keys():

57def encrypt_private_key(pin, usb_path):
58 global private_key
59 salt = get_random_bytes(16)
60 key = scrypt(pin.encode(), salt, 32, N=2**14, r=8, p=1)
61 cipher = AES.new(key, AES.MODE_GCM)
62 ciphertext, tag = cipher.encrypt_and_digest(private_key)
63
64 output_file = os.path.join(usb_path, "encrypted_private_key.bin")
65 try:
66 with open(output_file, "wb") as f:
67 for x in (salt, cipher.nonce, tag, ciphertext):
68 f.write(x)
69 return True
70 except Exception as e:
71 print(f"Błąd zapisu na pendrive: {e}")
72 return False
